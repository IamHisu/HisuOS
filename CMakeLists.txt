cmake_minimum_required(VERSION 3.16)

set(HISU_BOARD "zhengchen_154" CACHE STRING "HisuOS board profile")
set(HISU_BOARD_DIR "${CMAKE_CURRENT_SOURCE_DIR}/boards/${HISU_BOARD}")
if(NOT EXISTS "${HISU_BOARD_DIR}/board_config.h")
    message(FATAL_ERROR "Unknown HisuOS board: ${HISU_BOARD}")
endif()
set(IDF_TARGET "esp32s3" CACHE STRING "ESP-IDF target")
set(SDKCONFIG_DEFAULTS "sdkconfig.defaults;${HISU_BOARD_DIR}/sdkconfig.defaults")
include($ENV{IDF_PATH}/tools/cmake/project.cmake)
project(HisuOS VERSION 0.1.0)
